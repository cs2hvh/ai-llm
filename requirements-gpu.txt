# GPU-pod deps. Install on RunPod after base image, IN ADDITION TO requirements.txt.
#
# IMPORTANT: this file does NOT inherit the base deps from requirements.txt.
# A clean pod must install BOTH:
#   pip install -r requirements.txt        # keras, optax, orbax, pydantic, …
#   pip install -r requirements-gpu.txt    # overlays the CUDA build of JAX + torch
#
# Single-venv co-existence of JAX (FSDP training) + PyTorch (teacher audit).
#
# Why this exact combo (researched 2026-05-13 after pod whack-a-mole):
#   torch==2.7.1 pins every nvidia-*-cu12 with == constraints to the
#   CUDA 12.6 ABI set (cuDNN 9.5.1.17, cublas 12.6.4.1, nccl 2.26.2, ...).
#   jax[cuda12]==0.4.38 accepts those same versions via its >= ranges
#   (cuDNN >=9.1,<10; cublas >=12.1.3.1; nccl >=2.18.1; ...). So pip
#   resolves a single nvidia-* set that satisfies both stacks in ONE
#   transaction. No --force-reinstall, no --no-deps, no manual cuDNN
#   pin needed.
#
#   The CUDA 12.6 ABI is forward-compat with driver 12.8.x (current
#   RunPod default) and driver 12.9+ (future B200 pods).
#
# Sources for this choice:
#   - jax-cuda12-plugin 0.4.38 PyPI metadata (loose >= pins)
#   - torch 2.7.1 PyPI metadata (strict == CUDA 12.6 pins)
#   - apple/axlearn pyproject.toml (production JAX + torch combo)
#
# Anti-patterns we hit and the script now avoids:
#   - DO NOT bump JAX past 0.5.x without also bumping torch: JAX 0.6+
#     raises its cuDNN floor to 9.8 → collides with torch 2.7.1's
#     ==9.5.1.17. When that day comes, see apple/axlearn.
#   - DO NOT --upgrade nvidia-cudnn-cu12 after install. Pip pulls
#     9.22.x → CUDNN_STATUS_NOT_INITIALIZED on JAX 0.4.38.
#   - DO NOT install torch from whl/cu124 (= torch 2.6 with cuDNN
#     9.1.0.70). The default PyPI torch 2.7.1 (cu126 build) is the
#     answer.
#   - DO NOT --force-reinstall any single nvidia-* package. The
#     nvidia namespace package (PEP 420) breaks on interleaved
#     uninstall + reinstall.

torch==2.7.1
torchvision==0.22.1
jax[cuda12]==0.4.38

# FlashAttention via JAX-Pallas is built in; no separate dep needed.
# transformers + accelerate handled in the main install transaction
# by pod_launch_gpu.sh (so the single resolver pass settles everything).
